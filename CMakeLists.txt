cmake_minimum_required(VERSION 3.22)
project(lilt VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Keep local checkout paths out of diagnostic strings, including dependencies.
add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}/=")
include(FetchContent)
include(CTest)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Optimize for this CPU (local builds only)")
# Keep GGML_NATIVE=OFF portable; local builds can opt into native CPU tuning.
foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
  set(GGML_${feature} OFF CACHE BOOL "Enable ${feature} (local builds only)")
endforeach()
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper
  GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git
  GIT_TAG 7246b7311e089fe092c4abe7cfad5d0921f8be00
  GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(whisper)
add_library(lilt_engine src/engine.cpp src/text.cpp)
target_include_directories(lilt_engine PUBLIC src)
target_link_libraries(lilt_engine PUBLIC whisper PkgConfig::PULSE Threads::Threads)
target_compile_options(lilt_engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lilt src/main.cpp src/app.cpp)
target_link_libraries(lilt PRIVATE lilt_engine PkgConfig::GTK)
target_compile_definitions(lilt PRIVATE LILT_VERSION="${PROJECT_VERSION}")
target_compile_options(lilt PRIVATE -Wall -Wextra -Wpedantic)
# The installer copies this executable directly; keep build paths out of it.
set_target_properties(lilt PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "")
if(BUILD_TESTING)
  add_executable(engine_test tests/engine_test.cpp)
  target_link_libraries(engine_test PRIVATE lilt_engine)
  add_test(NAME engine COMMAND engine_test)
  find_program(LILT_GJS gjs)
  if(LILT_GJS)
    add_test(NAME extension_text
      COMMAND "${LILT_GJS}" -m "${CMAKE_CURRENT_SOURCE_DIR}/tests/text_test.js")
  endif()
endif()
